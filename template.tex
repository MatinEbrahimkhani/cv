\section{Publications and Conferences}

\textbf{PerSpaCor: Correcting Space and ZWNJ Errors in Persian Text with Transformer Models} \\
\textbf{Matin Ebrahimkhani}, Ebrahim Ansari\\
\textit{Proceedings of the 15th International Conference on Recent Advances in Natural Language Processing (RANLP 2025)} \\

\begin{itemize}
    \item \textbf{Novel Research:} Engineered and evaluated four original transformer-based classifier architectures—ReLU-Norm, DualStep DropNet, and SimplexNet—achieving a \textbf{State-of-the-Art macro F1-score of 97.26\%} (98.38\% with interactive correction).
    \item \textbf{Dataset Leadership:} Independently curated, preprocessed, and \textbf{manually annotated} a massive dataset of over \textbf{12.7 million words}, creating a new high-quality benchmark for Persian NLP.
    \item \textbf{Open Source Impact:} Released all SOTA models, code, and the annotated corpus to the research community and deployed a public web-based correction system to make the research accessible to end-users.
    \item \textbf{Paper:} \href{https://aclanthology.org/2025.ranlp-1.40/}{ACL Anthology -- 2025.ranlp-1.40}
    \item \textbf{Artifacts:} \href{https://huggingface.co/PerSpaCor}{Hugging Face/PerSpaCor} $|$ \href{https://github.com/MatinEbrahimkhani/PerSpaCor_components}{GitHub/PerSpaCor\_components}
\end{itemize}

\vspace{7em}

\textbf{Efficient Adaptation of Foundation Models: A Comparative Study of Parameter-Efficient Fine-Tuning Methods for Domain-Specific Tasks} \\
\textbf{Matin Ebrahimkhani} Mina Sirinchi, Mo Nasher, Mobin Ebrahimkhani \\
\textit{(In Progress)} \\
Investigating parameter-efficient fine-tuning techniques (LoRA, QLoRA, Adapters, Prefix Tuning, and full fine-tuning) across multiple NLP tasks (text classification, question answering, sequence labeling, text generation). The study will evaluate accuracy/F1 (for classification/QA/labeling), BLEU/ROUGE (for generation), and resource metrics (training time/GPU memory, inference latency, and number of trained parameters) to compare performance–efficiency trade-offs. Planned code and scripts will be released for reproducibility.
